⚙️ 3. SQL MIGRATION CHO SUPABASE DASHBOARD

CREATE TABLE IF NOT EXISTS public.leaderboard (
    id BIGSERIAL PRIMARY KEY,
    gamertag TEXT NOT NULL,
    score INTEGER NOT NULL DEFAULT 0,
    wave INTEGER NOT NULL DEFAULT 1,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX IF NOT EXISTS idx_leaderboard_score_desc ON public.leaderboard (score DESC, wave DESC);
ALTER TABLE public.leaderboard ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow anon public read" ON public.leaderboard FOR SELECT TO anon, authenticated USING (true);
CREATE POLICY "Allow anon insert scores" ON public.leaderboard FOR INSERT TO anon, authenticated WITH CHECK (true);
